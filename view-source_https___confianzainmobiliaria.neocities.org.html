// Lee el archivo y, si el contenido normalizado resulta con menos de 2 líneas
// (típico en Android cuando el proveedor de almacenamiento aún no terminó de
// sincronizar una descarga recién creada, ej. "export (7).csv"), reintenta
// automáticamente un par de veces con una breve espera antes de declarar error.
function leerArchivo(file, intento) {
const MAX_INTENTOS = 3; // 1 lectura inicial + 2 reintentos
const reader = new FileReader();

reader.onload = function(e) {
let content = e.target.result;
if (typeof content !== 'string') content = '';

// Quitar BOM y caracteres nulos sueltos que a veces inserta Android al final del stream
if (content.charCodeAt(0) === 0xFEFF) content = content.substring(1);
content = content.replace(/\u0000/g, '');

const norm = content.replace(/\r\n/g, '\n').replace(/\r/g, '\n');
const linesNorm = norm.split('\n').filter(l => l.trim() !== '');

if (linesNorm.length < 2 && intento < MAX_INTENTOS - 1) {
// Contenido sospechosamente corto: probablemente la descarga aún no
// estaba completamente disponible. Reintentar tras una breve espera.
setTimeout(function() {
leerArchivo(file, intento + 1);
}, 350 * (intento + 1));
return;
}

if (linesNorm.length < 2) {
// Falló incluso tras reintentar: ahí sí mostramos el diagnóstico completo.
mostrarDiagnosticoFallo(file, content, intento + 1);
return;
}

parseCSV(norm, file.name, {
name: file.name,
size: file.size,
type: file.type,
charCount: content.length
});
};

reader.onerror = function() {
if (intento < MAX_INTENTOS - 1) {
setTimeout(function() {
leerArchivo(file, intento + 1);
}, 350 * (intento + 1));
return;
}
const err = reader.error;
const msg = [
'❌ Error de lectura tras ' + (intento + 1) + ' intento(s)',
'',
'── ARCHIVO ──',
'Nombre: ' + file.name,
'Tamaño: ' + file.size + ' bytes',
'MIME: ' + (file.type || '(vacío)'),
'',
'── ERROR FileReader ──',
'Nombre del error: ' + (err ? err.name : '(desconocido)'),
'Mensaje: ' + (err && err.message ? err.message : '(sin mensaje)')
].join('\n');
showDiagOverlay(msg, '✕ CERRAR');
showToast('Error al leer el archivo');
document.getElementById('csvInput').value = '';
};

reader.readAsText(file);
}

// Normaliza nombres de encabezado: quita tildes, pasa a minúsculas y recorta espacios.
function normalizarHeader(h) {

.toString()
.trim()
.toLowerCase()
.normalize('NFD')
.replace(/[\u0300-\u036f]/g, '');
}

// Alias aceptados por campo (todos pasados por normalizarHeader antes de comparar).
const CSV_ALIASES = {
direccion: ['direccion', 'dirección', 'address'],
comuna: ['comuna'],
condominio: ['condominio', 'edificio', 'proyecto'],
tipologia: ['tipologia', 'tipología', 'tipo'],
arriendo: ['arriendo total', 'arriendo', 'valor arriendo', 'precio', 'renta'],

unidad: ['unidad', 'depto', 'numero depto', 'número depto']
};

// Dado el mapa de índices de columna (col) y el array de columnas de la fila,
// devuelve el valor de la primera columna cuyo nombre coincide con algún alias del campo.
function getCampo(columns, col, campo) {
const aliases = CSV_ALIASES[campo] || [];
for (const alias of aliases) {
const idx = col[alias];
if (idx !== undefined && columns[idx] !== undefined) {
return columns[idx];
}
}
return '';
}

function parseCSV(content, filename, fileInfo) {

'MIME: ' + (info.type || '(vacío)'),
'Caracteres leídos (FileReader): ' + (info.charCount !== undefined ? info.charCount : '(desconocido)'),
'',
'── LÍNEAS DETECTADAS ──',
'No vacías tras normalizar: ' + lines.length,
'',
'── PRIMER BLOQUE DE CONTENIDO ──',
JSON.stringify(content.substring(0, 300)),
'',
'── MOTIVO EXACTO DEL RECHAZO ──',
'lines.length (' + lines.length + ') < 2 en parseCSV()'
].join('\n');
showDiagOverlay(msg, '✕ CERRAR');
showToast('Archivo CSV vacío o inválido');
return;
}

// Autodetectar separador: punto y coma o coma
const firstLine = lines[0];
const sep = (firstLine.split(';').length >= firstLine.split(',').length) ? ';' : ',';

// Construye el mapa nombre de columna -> índice, leyendo el encabezado (fila 0).
const headers = firstLine.split(sep);
const col = {};
headers.forEach((h, i) => {
col[normalizarHeader(h)] = i;
});

allData = [];

for (let i = 1; i < lines.length; i++) {
const columns = lines[i].split(sep);
if (columns.length < 2) continue;

const direccion = (getCampo(columns, col, 'direccion') || '').trim();
const comuna = (getCampo(columns, col, 'comuna') || '').trim();
const condominio = (getCampo(columns, col, 'condominio') || '').trim();

const arriendo = parseFloat(getCampo(columns, col, 'arriendo')) || 0;

const m2Depto = parseFloat(getCampo(columns, col, 'm2Depto')) || 0;
const m2Terraza = parseFloat(getCampo(columns, col, 'm2Terraza')) || 0;
const unidad = (getCampo(columns, col, 'unidad') || '').trim();

// Si no se pudo resolver una dirección para la fila, se omite (fila inválida/vacía).
if (!direccion) continue;

gastoComun: Math.round(gastoComun),
m2Depto,
m2Terraza,
unidad,
balcon

if (dtLabel) dtLabel.textContent = `${info.cantidadPropiedades} propiedades · ${fechaStr}`;
}

function populateFilters() {
const comunas = [...new Set(allData.map(d => d.comuna).filter(c => c))].sort();
const tipologias = [...new Set(allData.map(d => d.tipologia).filter(t => t))].sort();

const comunaSelect = document.getElementById('filterComuna');
const tipologiaSelect = document.getElementById('filterTipologia');

comunaSelect.innerHTML = '<option value="">Todas</option>';
tipologiaSelect.innerHTML = '<option value="">Todas</option>';

comunas.forEach(c => {

comunaSelect.appendChild(option);
});

tipologias.forEach(t => {
const option = document.createElement('option');

option.textContent = t;
tipologiaSelect.appendChild(option);
});
}

function applyFilters() {
const comuna = document.getElementById('filterComuna').value;
const tipologia = document.getElementById('filterTipologia').value;
const balcon = document.getElementById('filterBalcon').value;
const precioMin = parseFloat(document.getElementById('filterPrecioMin').value) || 0;
const precioMax = parseFloat(document.getElementById('filterPrecioMax').value) || Infinity;

filteredData = allData.filter(item => {
if (comuna && item.comuna !== comuna) return false;
if (tipologia && item.tipologia !== tipologia) return false;
if (balcon && item.balcon !== balcon) return false;
if (item.arriendo < precioMin) return false;
if (item.arriendo > precioMax) return false;
return true;
});

filteredData.sort((a, b) => {
let comparison = 0;
if (currentSortField === 'price') {
comparison = a.arriendo - b.arriendo;
}
if (currentSortField === 'name') {
comparison = a.condominio.localeCompare(b.condominio);
}
return currentSortDirection === 'asc' ? comparison : -comparison;
});

const terminoBusqueda = normalizarTexto(document.getElementById('buscadorUltra').value);
if (terminoBusqueda) {
displayData = filteredData.filter(item => {
const textoBusqueda = (item.direccion + ' ' + item.unidad + ' ' + item.comuna + ' ' + item.condominio).toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
return textoBusqueda.includes(terminoBusqueda);
});
} else {
displayData = [...filteredData];
}
renderProperties();
}

function copyToClipboard(text, message = 'Copiado ✓') {
if (navigator.clipboard && window.isSecureContext) {
navigator.clipboard.writeText(text).then(() => showToast(message));
} else {
const textarea = document.createElement('textarea');
textarea.value = text;
textarea.style.position = 'fixed';
textarea.style.opacity = '0';
document.body.appendChild(textarea);
textarea.select();
try {
document.execCommand('copy');
showToast(message);
} catch (err) {
showToast('Error al copiar');
}
document.body.removeChild(textarea);
}
}

btn.classList.add('visible');

btn.classList.remove('visible');
}
}

})();
